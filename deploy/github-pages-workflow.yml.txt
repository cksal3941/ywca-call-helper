name: Deploy to GitHub Pages

# main 에 push 되면 프론트엔드를 빌드해 GitHub Pages 로 배포.
# (백엔드는 Pages 에서 실행되지 않음 — 과정 데이터는 번들 스냅샷, 메모는 브라우저 localStorage)

on:
  push:
    branches: [main]
  workflow_dispatch: # 수동 실행 버튼

permissions:
  contents: read
  pages: write
  id-token: write

# 동시 배포 1개만
concurrency:
  group: pages
  cancel-in-progress: true

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: npm

      - run: npm ci

      - name: Build (Pages base path)
        run: npm run build
        env:
          # 저장소 이름 하위경로로 빌드 (https://<user>.github.io/ywca-call-helper/)
          GHP_BASE: /ywca-call-helper/

      - name: SPA 404 fallback (index.html 복사)
        run: cp dist/index.html dist/404.html

      - uses: actions/upload-pages-artifact@v3
        with:
          path: dist

  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - id: deployment
        uses: actions/deploy-pages@v4
